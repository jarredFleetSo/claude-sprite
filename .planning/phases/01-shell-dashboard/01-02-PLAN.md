export interface AppConfig {

  org: string

  theme?: 'light' | 'dark' | 'system'

export interface SpriteAPI {

  saveConfig: (cfg: Partial<AppConfig>) => Promise<void>
  lifecycle: (sprite: string, org: string, action: 'start' | 'stop' | 'destroy' | 'create') => Promise<{ success: boolean; error?: string }>
  runSpriteLogin: () => Promise<{ success: boolean; error?: string }>
  onLifecycleProgress: (cb: (msg: string) => void) => () => void
}
```

From src/preload/index.ts (IPC channels available):
- window.spriteAPI.loadConfig() -> config:load -> returns AppConfig | null
- window.spriteAPI.saveConfig(cfg) -> config:save -> persists to electron-store
- window.spriteAPI.runSpriteLogin() -> sprite:login -> spawns `sprite login` (opens browser for OAuth, waits for completion, reloads config)

From src/main/ipc/setup.ts (sprite:login handler):
- Spawns `sprite login` with stdio: 'inherit' -- opens browser automatically
- On exit code 0: reloads config from disk (sprite login writes to ~/.sprites/ keyring)
- Returns { success: true, config } on success, { success: false, error } on failure

From src/main/config-store.ts:
- loadConfig() checks electron-store first, then auto-imports from ~/.config/cs/config.toml
- Returns { ...config, autoImported: true } if imported from cs config
- Returns null if no config found anywhere

   import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query'
   import type { AppConfig } from '../lib/sprite-types'

   export function useConfig() {
     return useQuery({
       queryKey: ['config'],
       queryFn: () => window.spriteAPI.loadConfig(),
       staleTime: Infinity,  // Config doesn't change unless user saves
     })
   }

   export function useSaveConfig() {
     const queryClient = useQueryClient()
     return useMutation({
       mutationFn: (cfg: Partial<AppConfig>) => window.spriteAPI.saveConfig(cfg),
       onSuccess: () => queryClient.invalidateQueries({ queryKey: ['config'] }),
     })
   }
   ```

   import { Toaster } from '@/components/ui/sonner'
   import { useConfig } from './hooks/useConfig'

   const queryClient = new QueryClient()

   function AppContent() {
     const { data: config, isLoading } = useConfig()

     if (isLoading) {

         <div className="animate-spin h-8 w-8 border-2 border-primary border-t-transparent rounded-full" />
